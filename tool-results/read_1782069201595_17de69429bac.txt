   370→   370→      onOpenChange(false);
   371→   371→      resetForm();
   372→   372→    },
   373→   373→    onError: (err: Error) => {
   374→   374→      toast.error(err.message || 'Failed to create reflection');
   375→   375→    },
   376→   376→  });

   378→   378→  const resetForm = () => {
   379→   379→    setExamId('');
   380→   380→    setDifficulty('');

   384→   384→    setWhatWentWrong('');
   385→   385→    setBiggestLesson('');
   386→   386→    setActionPlan('');

   389→   389→  };

   391→   391→  const handleSubmit = () => {
   392→   392→    const result = reflectionFormSchema.safeParse({
   393→   393→      examId,
   394→   394→      difficulty: difficulty || undefined,
   395→   395→      confidence,
   396→   396→      emotionalState: emotionalState || undefined,
   397→   397→      whatWentWell: whatWentWell || '',
   398→   398→      whatWentWrong: whatWentWrong || '',

   400→   400→      actionPlan: actionPlan || undefined,
   401→   401→      targetScore: targetScore ? Number(targetScore) : undefined,
   402→   402→    });
   403→   403→    if (!result.success) {
   404→   404→      const errors: Record<string, string> = {};
   405→   405→      const fieldErrors = result.error.flatten().fieldErrors;
   406→   406→      for (const [key, messages] of Object.entries(fieldErrors)) {
   407→   407→        if (messages && messages.length > 0) {
   408→   408→          errors[key] = messages[0];
   409→   409→        }
   410→   410→      }
   411→   411→      setFormErrors(errors);
   412→   412→      return;
   413→   413→    }
   414→   414→    setFormErrors({});
   415→   415→    createMutation.mutate({
   416→   416→      examId,
   417→   417→      examName: selectedExam?.name ?? '',
   418→   418→      examDate: selectedExam?.examDate ?? null,
   419→   419→      score: selectedExam?.score ?? null,
   420→   420→      cutoff: selectedExam?.cutoff ?? null,
   421→   421→      gap: selectedExam?.cutoffGap ?? null,
   422→   422→      result: selectedExam?.result ?? null,
   423→   423→      difficulty: difficulty || null,
   424→   424→      confidence,
   425→   425→      emotionalState: emotionalState || null,
   426→   426→      whatWentWrong: whatWentWrong || null,
   427→   427→      whatWentWell: whatWentWell || null,
   428→   428→      biggestLesson: biggestLesson || null,
   429→   429→      actionPlan: actionPlan || null,
   430→   430→      targetScore: targetScore ? Number(targetScore) : null,
   431→   431→    });
   432→   432→  };
   433→   433→

   446→   446→      type="button"
   447→   447→      onClick={onClick}
   448→   448→      className={cn(
   449→   449→        'rounded-full border px-3 py-1.5 text-xs font-medium transition-all duration-200',
