    73→function formatDate(dateStr: string): string {
    74→  const d = new Date(dateStr);
    75→  return d.toLocaleDateString('en-IN', {
    76→    day: 'numeric',
    77→    month: 'short',
    78→    year: 'numeric',
    79→  });
    80→}
    81→
    82→function getRelativeDate(dateStr: string): string {
    83→  try {
    84→    return formatDistanceToNow(parseISO(dateStr), { addSuffix: true });
    85→  } catch {
    86→    return formatDate(dateStr);
    87→  }
    88→}

    90→function getCountdownLabel(days: number | null): {
    91→  label: string;
    92→  className: string;
    93→  urgent: boolean;
    94→} {
    95→  if (days === null || days === undefined)
    96→    return { label: '—', className: 'text-muted-foreground', urgent: false };
    97→  if (days === 0) return { label: 'Today!', className: 'text-red-600 font-bold', urgent: true };
    98→  if (days === 1) return { label: 'Tomorrow', className: 'text-amber-600 font-semibold', urgent: true };
    99→  if (days <= 7) return { label: `${days} days left`, className: 'text-red-600 font-semibold', urgent: true };
   100→  if (days <= 30) return { label: `${days} days left`, className: 'text-amber-600 font-medium', urgent: false };
   101→  return { label: `${days} days left`, className: 'text-emerald-600', urgent: false };
   102→}

   153→function ApplicationProgress({ status }: { status: string }) {
   154→  const steps = ['not-started', 'in-progress', 'completed'];
   155→  const currentIndex = steps.indexOf(status);
   156→  const labels = ['Not Started', 'In Progress', 'Completed'];
   157→
   158→  return (
   159→    <div className="flex items-center gap-1.5">
   160→      {steps.map((step, idx) => (
   161→        <div key={step} className="flex items-center gap-1.5">
   162→          <div className="flex flex-col items-center gap-1">
   163→            <div
   164→              className={cn(
   165→                'size-6 rounded-full flex items-center justify-center transition-all',
   166→                idx < currentIndex
   167→                  ? 'bg-emerald-500 text-white'
   168→                  : idx === currentIndex
   169→                  ? 'bg-amber-500 text-white'
   170→                  : 'bg-muted text-muted-foreground'
   171→              )}
   172→            >
   173→              {idx < currentIndex ? (
   174→                <CheckCircle2 className="size-3.5" />
   175→              ) : (
   176→                <span className="text-[10px] font-bold">{idx + 1}</span>
   177→              )}
   178→            </div>
   179→            <span className={cn(
   180→              'text-[9px] leading-tight',
   181→              idx === currentIndex ? 'font-medium text-foreground' : 'text-muted-foreground'
   182→            )}>
   183→              {labels[idx]}
   184→            </span>
   185→          </div>
   186→          {idx < steps.length - 1 && (
   187→            <div className={cn(
   188→              'h-0.5 w-6 rounded-full mb-4',
   189→              idx < currentIndex ? 'bg-emerald-400' : 'bg-muted'
   190→            )} />
   191→          )}
   192→        </div>
   193→      ))}
   194→    </div>
   195→  );
   196→}

   275→// ─── Add/Edit Upcoming Exam Dialog ─────────────────────────
   276→function UpcomingExamDialog({
   277→  open,
   278→  onOpenChange,
   279→  editExam,
   280→}: {
   281→  open: boolean;
   282→  onOpenChange: (open: boolean) => void;
   283→  editExam: UpcomingExam | null;
   284→}) {
   285→  const queryClient = useQueryClient();
   286→  const isEdit = editExam !== null;
   287→
   288→  const [form, setForm] = useState(() =>
   289→    editExam
   290→      ? {
   291→          name: editExam.name,
   292→          org: editExam.org ?? '',
   293→          priority: editExam.priority,
   294→          examDate: new Date(editExam.examDate).toISOString().split('T')[0],
   295→        }
   296→      : { name: '', org: '', priority: 'Medium', examDate: '' },
   297→  );
   298→
   299→  const createMutation = useMutation({
   300→    mutationFn: () =>
   301→      createUpcomingExam({
   302→        name: form.name,
   303→        org: form.org || undefined,
   304→        priority: form.priority,
   305→        examDate: form.examDate,
   306→      }),
   307→    onSuccess: () => {
   308→      toast.success('Upcoming exam added!');
   309→      queryClient.invalidateQueries({ queryKey: ['upcoming-exams'] });
   310→      onOpenChange(false);
   311→    },
   312→    onError: () => {
   313→      toast.error('Failed to save exam. Please try again.');
   314→    },
   315→  });
   316→
   317→  const updateMutation = useMutation({
   318→    mutationFn: () =>
   319→      updateUpcomingExam(editExam!.id, {
   320→        name: form.name,
   321→        org: form.org || undefined,
   322→        priority: form.priority,
   323→        examDate: form.examDate,
   324→      }),
   325→    onSuccess: () => {
   326→      toast.success('Exam updated successfully!');
   327→      queryClient.invalidateQueries({ queryKey: ['upcoming-exams'] });
   328→      onOpenChange(false);
   329→    },
   330→    onError: () => {
   331→      toast.error('Failed to update exam. Please try again.');
   332→    },
   333→  });
   334→
   335→  const handleSubmit = () => {
   336→    if (isEdit) {
   337→      updateMutation.mutate();
   338→    } else {
   339→      createMutation.mutate();
   340→    }
   341→  };
   342→
   343→  const isPending = createMutation.isPending || updateMutation.isPending;
   344→
   345→  return (
   346→    <Dialog open={open} onOpenChange={onOpenChange}>
   347→      <DialogContent className="sm:max-w-md">
   348→        <DialogHeader>
   349→          <DialogTitle>
   350→            {isEdit ? 'Edit Upcoming Exam' : 'Add Upcoming Exam'}
   351→          </DialogTitle>
   352→          <DialogDescription>
   353→            {isEdit ? 'Update exam details' : 'Track an upcoming exam to stay prepared'}
   354→          </DialogDescription>
   355→        </DialogHeader>
   356→        <div className="grid gap-4 py-2">
   357→          <div className="grid gap-2">
   358→            <Label htmlFor="ue-name">Exam Name *</Label>
   359→            <Input
   360→              id="ue-name"
   361→              placeholder="e.g. IBPS PO 2025"
   362→              value={form.name}
   363→              onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   364→            />
   365→          </div>
   366→          <div className="grid gap-2">
   367→            <Label htmlFor="ue-org">Organization (optional)</Label>
   368→            <Input
   369→              id="ue-org"
   370→              placeholder="e.g. IBPS"
   371→              value={form.org}
   372→              onChange={(e) => setForm((f) => ({ ...f, org: e.target.value }))}
   373→            />
   374→          </div>
   375→          <div className="grid gap-2">
   376→            <Label>Priority</Label>
   377→            <Select
   378→              value={form.priority}
   379→              onValueChange={(v) => setForm((f) => ({ ...f, priority: v }))}
   380→            >
   381→              <SelectTrigger>
   382→                <SelectValue />
   383→              </SelectTrigger>
   384→              <SelectContent>
   385→                <SelectItem value="High">High</SelectItem>
   386→                <SelectItem value="Medium">Medium</SelectItem>
   387→                <SelectItem value="Low">Low</SelectItem>
   388→              </SelectContent>
   389→            </Select>
   390→          </div>
   391→          <div className="grid gap-2">
   392→            <Label htmlFor="ue-date">Exam Date *</Label>
   393→            <Input
   394→              id="ue-date"
   395→              type="date"
   396→              value={form.examDate}
   397→              onChange={(e) => setForm((f) => ({ ...f, examDate: e.target.value }))}
   398→            />
   399→          </div>
   400→        </div>
   401→        <DialogFooter>
   402→          <DialogClose asChild>
   403→            <Button variant="outline">Cancel</Button>
   404→          </DialogClose>
   405→          <Button
   406→            onClick={handleSubmit}
   407→            disabled={isPending || !form.name || !form.examDate}
   408→            className="bg-emerald-600 text-white hover:bg-emerald-700"
   409→          >
   410→            {isPending ? (
   411→              <Loader2 className="mr-2 size-4 animate-spin" />
   412→            ) : null}
   413→            {isEdit ? 'Update' : 'Add Exam'}
   414→          </Button>
   415→        </DialogFooter>
   416→      </DialogContent>
   417→    </Dialog>
   418→  );
   419→}
