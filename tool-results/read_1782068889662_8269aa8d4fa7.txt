    76→  const h = new Date().getHours();
    77→  if (h < 12) return { text: 'Good morning', icon: Sun };

    91→function formatCountdown(days: number | null): string {
    92→  if (days === null || days === undefined) return '—';
    93→  if (days === 0) return 'Today!';
    94→  if (days === 1) return 'Tomorrow';
    95→  return `${days} days left`;
    96→}

   115→function getWeekStreakDots(): boolean[] {
   116→  // Simulated 7-day streak: last 7 days (today = last)
   117→  const today = new Date();
   118→  const dots: boolean[] = [];
   119→  for (let i = 6; i >= 0; i--) {
   120→    const d = new Date(today);
   121→    d.setDate(d.getDate() - i);
   122→    // Pseudo-random based on day — always mark today and 4 others as active
   123→    const dayHash = (d.getDate() * 7 + d.getMonth() * 31) % 10;
   124→    dots.push(i === 0 || dayHash < 5);
   125→  }
   126→  return dots;
   127→}

   155→function getCountdown(examDate: string): { days: number; hours: number; isPast: boolean } {
   156→  const target = new Date(examDate);
   157→  const now = new Date();
   158→  const diff = target.getTime() - now.getTime();
   159→  if (diff <= 0) return { days: 0, hours: 0, isPast: true };
   160→  return {
   161→    days: Math.floor(diff / 86400000),
   162→    hours: Math.floor((diff % 86400000) / 3600000),
   163→    isPast: false,
   164→  };
   165→}
