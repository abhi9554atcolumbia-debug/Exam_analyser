    90→    90→function groupExamsByYear(exams: Exam[]): Record<string, Exam[]> {
    91→    91→  const groups: Record<string, Exam[]> = {};
    92→    92→  for (const exam of exams) {
    93→    93→    const year = new Date(exam.examDate).getFullYear().toString();
    94→    94→    if (!groups[year]) groups[year] = [];
    95→    95→    groups[year].push(exam);
    96→    96→  }
    97→    97→  return Object.fromEntries(
    98→    98→    Object.entries(groups).sort(([a], [b]) => Number(b) - Number(a)),
    99→    99→  );
   100→   100→}
