   129→   129→function groupExamsByYear(exams: Exam[]): Record<string, Exam[]> {
   130→   130→  const groups: Record<string, Exam[]> = {};
   131→   131→  for (const exam of exams) {
   132→   132→    const year = new Date(exam.examDate).getFullYear().toString();
   133→   133→    if (!groups[year]) groups[year] = [];
   134→   134→    groups[year].push(exam);
   135→   135→  }
   136→   136→  return Object.fromEntries(
   137→   137→    Object.entries(groups).sort(([a], [b]) => Number(b) - Number(a)),
   138→   138→  );
   139→   139→}

   361→   361→function AddExamDialog({
   362→   362→  open,
   363→   363→  onOpenChange,
   364→   364→}: {
   365→   365→  open: boolean;
   366→   366→  onOpenChange: (open: boolean) => void;
   367→   367→}) {
   368→   368→  const queryClient = useQueryClient();
   369→   369→  const [form, setForm] = useState<ExamFormState>({ ...EMPTY_FORM });
   370→   370→
   371→   371→  const mutation = useMutation({
   372→   372→    mutationFn: () =>
   373→   373→      createExam({
   374→   374→        name: form.name,
   375→   375→        org: form.org || undefined,
   376→   376→        category: form.category,
   377→   377→        stage: form.stage,
   378→   378→        examDate: form.examDate,
   379→   379→        score: Number(form.score),
   380→   380→        maxScore: Number(form.maxScore),
   381→   381→        cutoff: Number(form.cutoff),
   382→   382→        result: form.result,
   383→   383→        sectionalScores: form.sections
   384→   384→          .filter((s) => s.section)
   385→   385→          .map((s) => ({
   386→   386→            section: s.section,
   387→   387→            score: Number(s.score),
   388→   388→            max: Number(s.max),
   389→   389→          })),
   390→   390→      }),
   391→   391→    onSuccess: () => {
   392→   392→      toast.success('Exam added successfully!');
   393→   393→      queryClient.invalidateQueries({ queryKey: ['exams'] });
   394→   394→      queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   395→   395→      onOpenChange(false);
   396→   396→      setForm({ ...EMPTY_FORM });
   397→   397→    },
   398→   398→    onError: () => {
   399→   399→      toast.error('Failed to add exam. Please try again.');
   400→   400→    },
   401→   401→  });
   402→   402→
   403→   403→  return (
   404→   404→    <Dialog open={open} onOpenChange={onOpenChange}>
   405→   405→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   406→   406→        <DialogHeader>
   407→   407→          <DialogTitle>Add Exam Result</DialogTitle>
   408→   408→        </DialogHeader>
   409→   409→        <ExamFormFields form={form} setForm={setForm} />
   410→   410→        <DialogFooter>
   411→   411→          <DialogClose asChild>
   412→   412→            <Button variant="outline">Cancel</Button>
   413→   413→          </DialogClose>
   414→   414→          <Button
   415→   415→            onClick={() => mutation.mutate()}
   416→   416→            disabled={mutation.isPending || !form.name || !form.examDate || !form.score}
   417→   417→            className="bg-emerald-600 text-white hover:bg-emerald-700"
   418→   418→          >
   419→   419→            {mutation.isPending ? 'Saving...' : 'Add Exam'}
   420→   420→          </Button>
   421→   421→        </DialogFooter>
   422→   422→      </DialogContent>
   423→   423→    </Dialog>
   424→   424→  );
   425→   425→}

   501→   501→  open,
   502→   502→  onOpenChange,
   503→   503→  exam,

   511→   511→  const mutation = useMutation({

   513→   513→      if (!exam) return Promise.reject('No exam');

   566→   566→function ScoreVisualization({
   567→   567→  score,
   568→   568→  maxScore,
   569→   569→  cutoff,

   579→   579→  return (
   580→   580→    <div className="space-y-3">

   588→   588→              passed
   589→   589→                ? 'bg-gradient-to-r from-emerald-500 to-teal-500 dark:from-emerald-400 dark:to-teal-400'

   597→   597→          className="absolute top-0 h-4 w-0.5 -translate-x-1/2 bg-red-600 dark:bg-red-400"
   598→   598→          style={{ left: `${Math.min(cutoffPct, 100)}%` }}

   600→   600→        />
   601→   601→        <div
   602→   602→          className="absolute -top-5 -translate-x-1/2 text-[10px] font-medium text-red-600 dark:text-red-400"

   605→   605→          cutoff

   611→   611→          {passed ? '✓ Above cutoff' : '✗ Below cutoff'}

   614→   614→      </div>
   615→   615→    </div>
   616→   616→  );
   617→   617→}

   976→   976→function exportExamsCSV(exams: Exam[]) {
   977→   977→  const headers = [
   978→   978→    'Name',
   979→   979→    'Organization',
   980→   980→    'Category',
   981→   981→    'Stage',
   982→   982→    'Date',
   983→   983→    'Score',
   984→   984→    'Max Score',
   985→   985→    'Cutoff',
   986→   986→    'Gap',
   987→   987→    'Result',
   988→   988→    'Attempt',
   989→   989→  ];
   990→   990→  const rows = exams.map((e) => [
   991→   991→    `"${e.name}"`,
   992→   992→    `"${e.org || ''}"`,
   993→   993→    e.category,
   994→   994→    e.stage,
   995→   995→    e.examDate,
   996→   996→    e.score,
   997→   997→    e.maxScore,
   998→   998→    e.cutoff,
   999→   999→    e.cutoffGap ?? '',
  1000→  1000→    e.result,
  1001→  1001→    e.attempt,
  1002→  1002→  ]);
  1003→  1003→
  1004→  1004→  const csvContent = [headers.join(','), ...rows.map((r) => r.join(','))].join(
  1005→  1005→    '\n',
  1006→  1006→  );
  1007→  1007→  const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
  1008→  1008→  const url = URL.createObjectURL(blob);
  1009→  1009→  const link = document.createElement('a');
  1010→  1010→  link.href = url;
  1011→  1011→  link.download = `exam-history-${new Date().toISOString().split('T')[0]}.csv`;
  1012→  1012→  document.body.appendChild(link);
  1013→  1013→  link.click();
  1014→  1014→  document.body.removeChild(link);
  1015→  1015→  URL.revokeObjectURL(url);
  1016→  1016→}
