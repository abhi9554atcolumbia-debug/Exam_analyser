    56→function getDueCountdown(dueDate: string | null): { text: string; urgent: boolean } | null {
    57→  if (!dueDate) return null;
    58→  const days = differenceInDays(parseISO(dueDate), new Date());
    59→  if (days < 0) return { text: `Overdue by ${Math.abs(days)} day${Math.abs(days) !== 1 ? 's' : ''}`, urgent: true };
    60→  if (days === 0) return { text: 'Due today', urgent: true };
    61→  if (days === 1) return { text: 'Due tomorrow', urgent: true };
    62→  if (days <= 7) return { text: `Due in ${days} days`, urgent: true };
    63→  return { text: `Due in ${days} days`, urgent: false };
    64→}

    85→/* ── Completion Rate Ring ──────────────────────────────────── */
    86→function CompletionRateRing({ rate, total }: { rate: number; total: number }) {
    87→  const radius = 40;
    88→  const circumference = 2 * Math.PI * radius;
    89→  const offset = circumference - (rate / 100) * circumference;
    90→
    91→  return (
    92→    <div className="flex flex-col items-center justify-center">
    93→      <div className="relative">
    94→        <svg width="110" height="110" className="-rotate-90">
    95→          <circle
    96→            cx="55" cy="55" r={radius}
    97→            fill="none"
    98→            strokeWidth="8"
    99→            className="stroke-muted"
   100→          />
   101→          <circle
   102→            cx="55" cy="55" r={radius}
   103→            fill="none"
   104→            strokeWidth="8"
   105→            strokeLinecap="round"
   106→            className="stroke-emerald-500 transition-all duration-700"
   107→            strokeDasharray={circumference}
   108→            strokeDashoffset={offset}
   109→          />
   110→        </svg>
   111→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   112→          <span className="text-2xl font-extrabold text-foreground">{rate}%</span>
   113→          <span className="text-[10px] text-muted-foreground">completed</span>
   114→        </div>
   115→      </div>
   116→      <p className="mt-2 text-xs text-muted-foreground">{total} total goals</p>
   117→    </div>
   118→  );
   119→}
