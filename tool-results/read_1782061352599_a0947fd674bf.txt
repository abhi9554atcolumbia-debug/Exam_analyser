   115→// ── Utilities ────────────────────────────────────────────────
   116→function formatFileSize(bytes: number | null): string {
   117→  if (bytes == null) return '—';
   118→  if (bytes === 0) return '0 B';
   119→  const units = ['B', 'KB', 'MB', 'GB'];
   120→  const k = 1024;
   121→  const i = Math.floor(Math.log(bytes) / Math.log(k));
   122→  const size = bytes / Math.pow(k, i);
   123→  return `${size.toFixed(i === 0 ? 0 : 1)} ${units[i]}`;
   124→}
   125→
   126→function formatDate(dateStr: string): string {
   127→  try {
   128→    return new Date(dateStr).toLocaleDateString('en-IN', {
   129→      day: '2-digit',
   130→      month: 'short',
   131→      year: 'numeric',
   132→    });
   133→  } catch {
   134→    return dateStr;
   135→  }
   136→}
   137→
   138→function getCategoryIcon(category: string): string {

   140→}
