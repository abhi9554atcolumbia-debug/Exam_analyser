   120→function formatDate(dateStr: string): string {
   121→  const d = new Date(dateStr);
   122→  return d.toLocaleDateString('en-IN', {
   123→    day: 'numeric',
   124→    month: 'short',
   125→    year: 'numeric',
   126→  });
   127→}

   129→function groupExamsByYear(exams: Exam[]): Record<string, Exam[]> {
   130→  const groups: Record<string, Exam[]> = {};
   131→  for (const exam of exams) {
   132→    const year = new Date(exam.examDate).getFullYear().toString();
   133→    if (!groups[year]) groups[year] = [];
   134→    groups[year].push(exam);
   135→  }
   136→  return Object.fromEntries(
   137→    Object.entries(groups).sort(([a], [b]) => Number(b) - Number(a)),
   138→  );
   139→}

   360→// ─── Add Exam Dialog ──────────────────────────────────────
   361→function AddExamDialog({
   362→  open,
   363→  onOpenChange,
   364→}: {
   365→  open: boolean;
   366→  onOpenChange: (open: boolean) => void;
   367→}) {
   368→  const queryClient = useQueryClient();
   369→  const [form, setForm] = useState<ExamFormState>({ ...EMPTY_FORM });
   370→
   371→  const mutation = useMutation({
   372→    mutationFn: () =>
   373→      createExam({
   374→        name: form.name,
   375→        org: form.org || undefined,
   376→        category: form.category,
   377→        stage: form.stage,
   378→        examDate: form.examDate,
   379→        score: Number(form.score),
   380→        maxScore: Number(form.maxScore),
   381→        cutoff: Number(form.cutoff),
   382→        result: form.result,
   383→        sectionalScores: form.sections
   384→          .filter((s) => s.section)
   385→          .map((s) => ({
   386→            section: s.section,
   387→            score: Number(s.score),
   388→            max: Number(s.max),
   389→          })),
   390→      }),
   391→    onSuccess: () => {
   392→      toast.success('Exam added successfully!');
   393→      queryClient.invalidateQueries({ queryKey: ['exams'] });
   394→      queryClient.invalidateQueries({ queryKey: ['dashboard'] });
   395→      onOpenChange(false);
   396→      setForm({ ...EMPTY_FORM });
   397→    },
   398→    onError: () => {
   399→      toast.error('Failed to add exam. Please try again.');
   400→    },
   401→  });
   402→
   403→  return (
   404→    <Dialog open={open} onOpenChange={onOpenChange}>
   405→      <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   406→        <DialogHeader>
   407→          <DialogTitle>Add Exam Result</DialogTitle>
   408→        </DialogHeader>
   409→        <ExamFormFields form={form} setForm={setForm} />
   410→        <DialogFooter>
   411→          <DialogClose asChild>
   412→            <Button variant="outline">Cancel</Button>
   413→          </DialogClose>
   414→          <Button
   415→            onClick={() => mutation.mutate()}
   416→            disabled={mutation.isPending || !form.name || !form.examDate || !form.score}
   417→            className="bg-emerald-600 text-white hover:bg-emerald-700"
   418→          >
   419→            {mutation.isPending ? 'Saving...' : 'Add Exam'}
   420→          </Button>
   421→        </DialogFooter>
   422→      </DialogContent>
   423→    </Dialog>
   424→  );
   425→}

   565→// ─── Score Visualization Bar ───────────────────────────────
   566→function ScoreVisualization({
   567→  score,
   568→  maxScore,
   569→  cutoff,
   570→}: {
   571→  score: number;
   572→  maxScore: number;
   573→  cutoff: number;
   574→}) {
   575→  const scorePct = maxScore > 0 ? Math.round((score / maxScore) * 100) : 0;
   576→  const cutoffPct = maxScore > 0 ? Math.round((cutoff / maxScore) * 100) : 0;
   577→  const passed = score >= cutoff;
   578→
   579→  return (
   580→    <div className="space-y-3">
   581→      <div className="relative">
   582→        {/* Track */}
   583→        <div className="h-4 w-full overflow-hidden rounded-full bg-muted">
   584→          {/* Score fill */}
   585→          <div
   586→            className={cn(
   587→              'h-full rounded-full transition-all duration-700 ease-out',
   588→              passed
   589→                ? 'bg-gradient-to-r from-emerald-500 to-teal-500 dark:from-emerald-400 dark:to-teal-400'
   590→                : 'bg-gradient-to-r from-amber-500 to-red-500 dark:from-amber-400 dark:to-red-400',
   591→            )}
   592→            style={{ width: `${Math.min(scorePct, 100)}%` }}
   593→          />
   594→        </div>
   595→        {/* Cutoff marker */}
   596→        <div
   597→          className="absolute top-0 h-4 w-0.5 -translate-x-1/2 bg-red-600 dark:bg-red-400"
   598→          style={{ left: `${Math.min(cutoffPct, 100)}%` }}
   599→          title={`Cutoff: ${cutoff}`}
   600→        />
   601→        <div
   602→          className="absolute -top-5 -translate-x-1/2 text-[10px] font-medium text-red-600 dark:text-red-400"
   603→          style={{ left: `${Math.min(cutoffPct, 100)}%` }}
   604→        >
   605→          cutoff

   975→// ─── CSV Export ────────────────────────────────────────────
   976→function exportExamsCSV(exams: Exam[]) {
   977→  const headers = [
   978→    'Name',
   979→    'Organization',
   980→    'Category',
   981→    'Stage',
   982→    'Date',
   983→    'Score',
   984→    'Max Score',
   985→    'Cutoff',
   986→    'Gap',
   987→    'Result',
   988→    'Attempt',
   989→  ];
   990→  const rows = exams.map((e) => [
   991→    `"${e.name}"`,
   992→    `"${e.org || ''}"`,
   993→    e.category,
   994→    e.stage,
   995→    e.examDate,
   996→    e.score,
   997→    e.maxScore,
   998→    e.cutoff,
   999→    e.cutoffGap ?? '',
  1000→    e.result,
  1001→    e.attempt,
  1002→  ]);
  1003→
  1004→  const csvContent = [headers.join(','), ...rows.map((r) => r.join(','))].join(
  1005→    '\n',
  1006→  );
  1007→  const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
  1008→  const url = URL.createObjectURL(blob);
  1009→  const link = document.createElement('a');
  1010→  link.href = url;
  1011→  link.download = `exam-history-${new Date().toISOString().split('T')[0]}.csv`;
  1012→  document.body.appendChild(link);
  1013→  link.click();
  1014→  document.body.removeChild(link);
  1015→  URL.revokeObjectURL(url);
  1016→}
