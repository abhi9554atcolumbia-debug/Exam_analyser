    81→function formatDate(dateStr: string): string {
    82→  const d = new Date(dateStr);
    83→  return d.toLocaleDateString('en-IN', {
    84→    day: 'numeric',
    85→    month: 'short',
    86→    year: 'numeric',
    87→  });

    90→function groupExamsByYear(exams: Exam[]): Record<string, Exam[]> {
    91→  const groups: Record<string, Exam[]> = {};
    92→  for (const exam of exams) {
    93→    const year = new Date(exam.examDate).getFullYear().toString();
    94→    if (!groups[year]) groups[year] = [];
    95→    groups[year].push(exam);
    96→  }
    97→  return Object.fromEntries(
    98→    Object.entries(groups).sort(([a], [b]) => Number(b) - Number(a)),
    99→  );
   100→}
