    58→const EMOTIONAL_OPTIONS = ['Confident', 'Anxious', 'Neutral', 'Frustrated'];

    60→  'Time Management',

    95→// ── Component ────────────────────────────────────────────────
    96→export default function ReflectionsPage() {
    97→  const { navigate, previousPage } = useNavigationStore();
    98→  const queryClient = useQueryClient();
    99→
   100→  // editing reflection id (passed via store or query param style)
   101→  const [editingId, setEditingId] = useState<string | null>(null);
   102→  const [selectedExamId, setSelectedExamId] = useState<string | null>(null);
   103→  const [manualExamData, setExamData] = useState<Exam | null>(null);
   104→
   105→  // Form state
   106→  const [difficulty, setDifficulty] = useState('');
   107→  const [confidence, setConfidence] = useState(50);
   108→  const [emotionalState, setEmotionalState] = useState('');
   109→  const [mistakeTags, setMistakeTags] = useState<string[]>([]);
   110→  const [strengthTags, setStrengthTags] = useState<string[]>([]);
   111→  const [whatWentWrong, setWhatWentWrong] = useState('');
   112→  const [whatWentWell, setWhatWentWell] = useState('');
   113→  const [biggestLesson, setBiggestLesson] = useState('');
   114→  const [actionPlan, setActionPlan] = useState('');
   115→  const [targetScore, setTargetScore] = useState('');
   116→  const [goalDescription, setGoalDescription] = useState('');
   117→  const [manualSections, setSectionReflections] = useState<SectionReflection[]>([]);
   118→  const [attachments, setAttachments] = useState<string[]>([]);
   119→  const [reminderType, setReminderType] = useState('');
   120→  const [reminderDate, setReminderDate] = useState('');
   121→
   122→  // ── Queries ─────────────────────────────────────────────
   123→  const { data: examOptions, isLoading: loadingOptions } = useQuery({
   124→    queryKey: ['reflection-exam-options'],
   125→    queryFn: getReflectionExamOptions,
   126→  });
   127→
   128→  const { data: reflections, isLoading: loadingReflections } = useQuery({
   129→    queryKey: ['reflections-list'],
   130→    queryFn: () => getReflections(),
   131→  });
   132→
   133→  const { data: selectedExam, isLoading: loadingExam } = useQuery({
   134→    queryKey: ['exam-for-reflection', selectedExamId],
   135→    queryFn: () => getExam(selectedExamId!),
   136→    enabled: !!selectedExamId,
   137→  });
   138→
   139→  // Derive examData from query result (avoids synchronous setState in effect)
   140→  const derivedExamData = selectedExam;
   141→  const derivedSections = selectedExam?.sectionalScores?.length
   142→    ? selectedExam.sectionalScores.map((s) => ({
   143→        section: s.section,
   144→        score: String(s.score ?? ''),
   145→        strength: '',
   146→        weakness: '',
   147→        actionPlan: '',
   148→      }))
   149→    : [];
   150→  // examData and sectionReflections prefer manually-set values (from loadReflection), else use query-derived
   151→  const examData = manualExamData || derivedExamData;
   152→  const sectionReflections = manualSections.length > 0 ? manualSections : derivedSections;
   153→
   154→  // ── Load existing reflection ────────────────────────────
   155→  const loadReflection = useCallback(
   156→    (r: Reflection) => {
   157→      setEditingId(r.id);
   158→      if (r.examId) {
   159→        setSelectedExamId(r.examId);
   160→        getExam(r.examId).then((exam) => {
   161→          setExamData(exam);
   162→          if (exam.sectionalScores?.length) {
   163→            const sectionMap = new Map(
   164→              r.sections?.map((s) => [s.section, s]) ?? []
   165→            );
   166→            setSectionReflections(
   167→              exam.sectionalScores.map((s) => {
   168→                const existing = sectionMap.get(s.section);
   169→                return {
   170→                  section: s.section,
   171→                  score: existing?.score != null ? String(existing.score) : String(s.score ?? ''),
   172→                  strength: existing?.strength ?? '',
   173→                  weakness: existing?.weakness ?? '',
   174→                  actionPlan: existing?.actionPlan ?? '',
   175→                };
   176→              })
   177→            );
   178→          }
   179→        });
   180→      }
   181→      setDifficulty(r.difficulty ?? '');
   182→      setConfidence(r.confidence ?? 50);
   183→      setEmotionalState(r.emotionalState ?? '');
   184→      setMistakeTags(r.mistakeTags ? r.mistakeTags.split(',').filter(Boolean) : []);
   185→      setStrengthTags(r.strengthTags ? r.strengthTags.split(',').filter(Boolean) : []);
   186→      setWhatWentWrong(r.whatWentWrong ?? '');
   187→      setWhatWentWell(r.whatWentWell ?? '');
   188→      setBiggestLesson(r.biggestLesson ?? '');
   189→      setActionPlan(r.actionPlan ?? '');
   190→      setTargetScore(r.targetScore != null ? String(r.targetScore) : '');
   191→      setGoalDescription(r.goalDescription ?? '');
   192→      setReminderType(r.reminderType ?? '');
   193→      setReminderDate(r.reminderDate ?? '');
   194→    },
   195→    []
   196→  );
   197→
   198→  // ── Mutations ───────────────────────────────────────────
   199→  const saveMutation = useMutation({
   200→    mutationFn: (data: Parameters<typeof createReflection>[0]) => {
   201→      if (editingId) return updateReflection(editingId, data);
   202→      return createReflection(data);
   203→    },
   204→    onSuccess: () => {
   205→      toast.success(editingId ? 'Reflection updated!' : 'Reflection saved!');
   206→      queryClient.invalidateQueries({ queryKey: ['reflections'] });
   207→      queryClient.invalidateQueries({ queryKey: ['reflection-exam-options'] });
   208→      queryClient.invalidateQueries({ queryKey: ['exams'] });
   209→      queryClient.invalidateQueries({ queryKey: ['analytics'] });
   210→      queryClient.invalidateQueries({ queryKey: ['weakness'] });
   211→      navigate('history');
   212→    },
   213→    onError: (err: Error) => {
   214→      toast.error(err.message || 'Failed to save reflection');
   215→    },
   216→  });
   217→
   218→  // ── Handlers ────────────────────────────────────────────
   219→  const toggleTag = (tag: string, list: string[], setter: (v: string[]) => void) => {
   220→    setter(list.includes(tag) ? list.filter((t) => t !== tag) : [...list, tag]);
   221→  };
   222→
   223→  const updateSection = (idx: number, field: keyof SectionReflection, value: string) => {
   224→    setSectionReflections((prev) =>
   225→      prev.map((s, i) => (i === idx ? { ...s, [field]: value } : s))
   226→    );
   227→  };
   228→
   229→  const handleSave = () => {
   230→    if (!selectedExamId) {
   231→      toast.error('Please select an exam');
   232→      return;
   233→    }
   234→    const payload: Parameters<typeof createReflection>[0] = {
   235→      examId: selectedExamId,
   236→      examName: examData?.name ?? '',
   237→      examDate: examData?.examDate ?? null,
   238→      score: examData?.score ?? null,
   239→      cutoff: examData?.cutoff ?? null,
   240→      gap: examData?.cutoffGap ?? null,
   241→      result: examData?.result ?? null,
   242→      difficulty: difficulty || null,
   243→      confidence,
   244→      emotionalState: emotionalState || null,
   245→      mistakeTags: mistakeTags.length ? mistakeTags.join(',') : null,
   246→      strengthTags: strengthTags.length ? strengthTags.join(',') : null,
   247→      whatWentWrong: whatWentWrong || null,
   248→      whatWentWell: whatWentWell || null,
   249→      biggestLesson: biggestLesson || null,
   250→      actionPlan: actionPlan || null,
   251→      targetScore: targetScore ? Number(targetScore) : null,
   252→      goalDescription: goalDescription || null,
   253→      reminderType: reminderType || null,
   254→      reminderDate: reminderDate || null,
   255→      sections: sectionReflections.map((s) => ({
   256→        section: s.section,
   257→        score: s.score ? Number(s.score) : undefined,
   258→        strength: s.strength || undefined,
   259→        weakness: s.weakness || undefined,
   260→        actionPlan: s.actionPlan || undefined,
   261→      })),
   262→    };
   263→    saveMutation.mutate(payload);
   264→  };
   265→
   266→  const handleCancel = () => {
   267→    resetForm();
   268→    if (previousPage && previousPage !== 'reflections') {
   269→      navigate(previousPage);
   270→    } else {
   271→      navigate('history');
   272→    }
   273→  };
   274→
   275→  const resetForm = () => {
   276→    setEditingId(null);
   277→    setSelectedExamId(null);
   278→    setExamData(null);
   279→    setDifficulty('');
   280→    setConfidence(50);
   281→    setEmotionalState('');
   282→    setMistakeTags([]);
   283→    setStrengthTags([]);
   284→    setWhatWentWrong('');
   285→    setWhatWentWell('');
   286→    setBiggestLesson('');
   287→    setActionPlan('');
   288→    setTargetScore('');
   289→    setGoalDescription('');
   290→    setSectionReflections([]);
   291→    setAttachments([]);
   292→    setReminderType('');
   293→    setReminderDate('');
   294→  };
   295→
   296→  const handleSelectExam = (examId: string) => {
   297→    if (editingId) return; // don't switch exam when editing
   298→    resetForm();
   299→    setSelectedExamId(examId);
   300→  };
   301→
   302→  const addAttachment = (name: string) => {
   303→    setAttachments((prev) => [...prev, name]);
   304→  };
   305→
   306→  const removeAttachment = (idx: number) => {
   307→    setAttachments((prev) => prev.filter((_, i) => i !== idx));
   308→  };
   309→
   310→  // ── Render helpers ──────────────────────────────────────
   311→  const ToggleButton = ({
   312→    label,
   313→    active,
   314→    onClick,
   315→  }: {
   316→    label: string;
   317→    active: boolean;
   318→    onClick: () => void;
   319→  }) => (
   320→    <button
   321→      type="button"
   322→      onClick={onClick}
   323→      className={`rounded-full border px-4 py-1.5 text-sm font-medium transition-colors ${
   324→        active
   325→          ? 'border-emerald-500 bg-emerald-500/15 text-emerald-700 dark:text-emerald-400'
   326→          : 'border-border bg-background text-muted-foreground hover:border-emerald-400 hover:text-emerald-600'
   327→      }`}
   328→    >
   329→      {label}
   330→    </button>
   331→  );
   332→
   333→  const resultColor = (r: string) => {
   334→    if (r === 'Qualified' || r === 'qualified') return 'text-emerald-600 dark:text-emerald-400';
   335→    if (r === 'Not Qualified' || r === 'not-qualified') return 'text-red-600 dark:text-red-400';
   336→    return 'text-muted-foreground';
   337→  };
   338→
   339→  // ── Loading state ───────────────────────────────────────
   340→  if (loadingOptions && loadingReflections) {
   341→    return (
   342→      <div className="space-y-6 p-4 md:p-6">
   343→        <Skeleton className="h-8 w-48" />
   344→        <Skeleton className="h-12 w-full" />
   345→        <Skeleton className="h-64 w-full" />
   346→        <Skeleton className="h-48 w-full" />
   347→      </div>
   348→    );
   349→  }
   350→
   351→  const existingReflections = reflections ?? [];
   352→
   353→  return (
   354→    <div className="space-y-6 p-4 md:p-6">
   355→      {/* ── Header ──────────────────────────────────────── */}
   356→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   357→        <div className="flex items-center gap-3">
   358→          <button
   359→            onClick={() => navigate(previousPage ?? 'history')}
   360→            className="rounded-lg p-1.5 text-muted-foreground hover:bg-muted hover:text-foreground transition-colors"
   361→          >
   362→            <ArrowLeft className="size-5" />
   363→          </button>
   364→          <div>
   365→            <h1 className="text-2xl font-bold tracking-tight">Exam Reflection</h1>
   366→            <p className="text-sm text-muted-foreground">
   367→              {editingId ? 'Edit your reflection' : 'Create a detailed reflection for an exam'}
   368→            </p>
   369→          </div>
   370→        </div>
   371→        <Button
   372→          variant="outline"
   373→          size="sm"
   374→          onClick={() => navigate('history')}
   375→          className="self-start sm:self-auto"
   376→        >
   377→          <ArrowLeft className="mr-1.5 size-3.5" />
   378→          Back to History
   379→        </Button>
   380→      </div>
   381→
   382→      {/* ── Existing reflections (when no exam selected) ── */}
   383→      {!selectedExamId && !editingId && (
   384→        <div className="space-y-4">
   385→          <Card>
   386→            <CardHeader className="pb-3">
   387→              <CardTitle className="text-base">Select Exam to Reflect On</CardTitle>
   388→            </CardHeader>
   389→            <CardContent>
   390→              <Select value={selectedExamId ?? ''} onValueChange={handleSelectExam}>
   391→                <SelectTrigger className="w-full">
   392→                  <SelectValue placeholder="Choose an exam..." />
   393→                </SelectTrigger>
   394→                <SelectContent>
   395→                  {examOptions?.map((e) => (
   396→                    <SelectItem key={e.id} value={e.id}>
   397→                      {e.name} — {e.examDate}
   398→                    </SelectItem>
   399→                  ))}
   400→                  {(!examOptions || examOptions.length === 0) && (
   401→                    <SelectItem value="__none" disabled>
   402→                      No exams available
   403→                    </SelectItem>
   404→                  )}
   405→                </SelectContent>
   406→              </Select>
   407→            </CardContent>
   408→          </Card>
   409→
   410→          {existingReflections.length > 0 && (
   411→            <Card>
   412→              <CardHeader className="pb-3">
   413→                <CardTitle className="text-base">Existing Reflections</CardTitle>
   414→              </CardHeader>
   415→              <CardContent>
   416→                <div className="space-y-2">
   417→                  {existingReflections.map((r) => (
   418→                    <button
   419→                      key={r.id}
   420→                      onClick={() => loadReflection(r)}
   421→                      className="flex w-full items-center justify-between rounded-lg border p-3 text-left hover:bg-muted/50 transition-colors"
   422→                    >
   423→                      <div>
   424→                        <p className="font-medium text-sm">{r.examName}</p>
   425→                        <p className="text-xs text-muted-foreground">
   426→                          {r.examDate} · Score: {r.score} · {r.difficulty}
   427→                        </p>
   428→                      </div>
   429→                      <Badge variant="secondary" className="text-xs shrink-0">
   430→                        {r.emotionalState}
   431→                      </Badge>
   432→                    </button>
   433→                  ))}
   434→                </div>
   435→              </CardContent>
   436→            </Card>
   437→          )}
   438→        </div>
   439→      )}
   440→
   441→      {/* ── Reflection Form ─────────────────────────────── */}
   442→      {selectedExamId && (
   443→        <div className="space-y-6">
   444→          {/* Exam Summary Card */}
   445→          <Card className="border-emerald-500/30 bg-emerald-500/5 dark:bg-emerald-500/10">
   446→            <CardContent className="p-4">
   447→              {loadingExam ? (
   448→                <div className="flex items-center gap-4">
   449→                  <Skeleton className="h-16 w-16 rounded-lg" />
   450→                  <div className="space-y-2">
   451→                    <Skeleton className="h-5 w-40" />
   452→                    <Skeleton className="h-4 w-24" />
   453→                  </div>
   454→                </div>
   455→              ) : examData ? (
   456→                <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   457→                  <div className="flex items-center gap-4">
   458→                    <div className="flex size-12 items-center justify-center rounded-lg bg-emerald-500/15 text-emerald-600 dark:text-emerald-400">
   459→                      <FileText className="size-6" />
   460→                    </div>
   461→                    <div>
   462→                      <h3 className="font-semibold">{examData.name}</h3>
   463→                      <p className="text-sm text-muted-foreground">
   464→                        {examData.examDate} · {examData.category} · Attempt #{examData.attempt}
   465→                      </p>
   466→                    </div>
   467→                  </div>
   468→                  <div className="flex items-center gap-4 text-sm">
   469→                    <div className="text-center">
   470→                      <p className="text-muted-foreground text-xs">Score</p>
   471→                      <p className="font-semibold">{examData.score}/{examData.maxScore}</p>
   472→                    </div>
   473→                    <div className="text-center">
   474→                      <p className="text-muted-foreground text-xs">Cutoff</p>
   475→                      <p className="font-semibold">{examData.cutoff}</p>
   476→                    </div>
   477→                    <div className="text-center">
   478→                      <p className="text-muted-foreground text-xs">Gap</p>
   479→                      <p className={`font-semibold ${examData.cutoffGap != null && examData.cutoffGap < 0 ? 'text-emerald-600 dark:text-emerald-400' : 'text-red-600 dark:text-red-400'}`}>
   480→                        {examData.cutoffGap != null ? `${examData.cutoffGap > 0 ? '+' : ''}${examData.cutoffGap}` : '—'}
   481→                      </p>
   482→                    </div>
   483→                    <Badge
   484→                      variant={examData.result === 'Qualified' || examData.result === 'qualified' ? 'default' : 'destructive'}
   485→                      className={examData.result === 'Qualified' || examData.result === 'qualified' ? 'bg-emerald-600 hover:bg-emerald-700 dark:bg-emerald-600' : ''}
   486→                    >
   487→                      {examData.result}
   488→                    </Badge>
   489→                  </div>
   490→                </div>
   491→              ) : null}
   492→            </CardContent>
   493→          </Card>
   494→
   495→          {/* 1. Exam Experience */}
   496→          <Card>
   497→            <CardHeader className="pb-3">
   498→              <CardTitle className="text-base">1. Exam Experience</CardTitle>
   499→            </CardHeader>
   500→            <CardContent className="space-y-5">
   501→              <div className="space-y-2">
   502→                <Label>Difficulty Level</Label>
   503→                <div className="flex flex-wrap gap-2">
   504→                  {DIFFICULTY_OPTIONS.map((d) => (
   505→                    <ToggleButton
   506→                      key={d}
   507→                      label={d}
   508→                      active={difficulty === d}
   509→                      onClick={() => setDifficulty(difficulty === d ? '' : d)}
   510→                    />
   511→                  ))}
   512→                </div>
   513→              </div>
   514→              <div className="space-y-3">
   515→                <div className="flex items-center justify-between">
   516→                  <Label>Confidence Level</Label>
   517→                  <span className="text-sm font-medium text-emerald-600 dark:text-emerald-400">
   518→                    {confidence}%
   519→                  </span>
   520→                </div>
   521→                <Slider
   522→                  value={[confidence]}
   523→                  onValueChange={(v) => setConfidence(v[0])}
   524→                  min={0}
   525→                  max={100}
   526→                  step={5}
   527→                  className="[&_[data-slot=slider-range]]:bg-emerald-500 [&_[data-slot=slider-thumb]]:border-emerald-500"
   528→                />
   529→                <div className="flex justify-between text-xs text-muted-foreground">
   530→                  <span>Not Confident</span>
   531→                  <span>Very Confident</span>
   532→                </div>
   533→              </div>
   534→              <div className="space-y-2">
   535→                <Label>Emotional State</Label>
   536→                <div className="flex flex-wrap gap-2">
   537→                  {EMOTIONAL_OPTIONS.map((e) => (
   538→                    <ToggleButton
   539→                      key={e}
   540→                      label={e}
   541→                      active={emotionalState === e}
   542→                      onClick={() => setEmotionalState(emotionalState === e ? '' : e)}
   543→                    />
   544→                  ))}
   545→                </div>
   546→              </div>
   547→            </CardContent>
   548→          </Card>
   549→
   550→          {/* 2. Mistake Tags */}
   551→          <Card>
   552→            <CardHeader className="pb-3">
   553→              <CardTitle className="text-base">2. Mistake Tags</CardTitle>
   554→              <p className="text-xs text-muted-foreground">Select the mistakes you made</p>
   555→            </CardHeader>
   556→            <CardContent>
