    67→function formatDate(dateStr: string): string {
    68→  const d = new Date(dateStr);
    69→  return d.toLocaleDateString('en-IN', {
    70→    day: 'numeric',
    71→    month: 'short',
    72→    year: 'numeric',
    73→  });
    74→}

    76→function groupExamsByYear(exams: Exam[]): Record<string, Exam[]> {
    77→  const groups: Record<string, Exam[]> = {};
    78→  for (const exam of exams) {
    79→    const year = new Date(exam.examDate).getFullYear().toString();
    80→    if (!groups[year]) groups[year] = [];
    81→    groups[year].push(exam);
    82→  }
    83→  return Object.fromEntries(
    84→    Object.entries(groups).sort(([a], [b]) => Number(b) - Number(a)),
    85→  );
    86→}

   586→        year: year !== 'all' ? year : undefined,
   587→        status: status !== 'all' ? status : undefined,
   588→      }),
   589→  });
   590→
   591→  const exams = data?.data ?? [];
   592→  const pagination = data?.pagination;
   593→  const hasMore = pagination ? page < pagination.pages : false;
   594→
   595→  const grouped = useMemo(() => groupExamsByYear(exams), [exams]);
   596→
   597→  // Stats computed from all loaded exams
   598→  const stats = useMemo(() => {
   599→    const total = exams.length;
   600→    const qualified = exams.filter((e) => e.result === 'Qualified').length;
   601→    const avgScore =
   602→      total > 0
   603→        ? Math.round(
   604→            (exams.reduce(
   605→              (s, e) => s + (e.score / e.maxScore) * 100,
   606→              0,
   607→            ) /
   608→              total) *
   609→              10,
   610→          ) / 10
   611→        : 0;
   612→    const bestScore =
   613→      total > 0
   614→        ? Math.round(
   615→            (Math.max(...exams.map((e) => e.score / e.maxScore)) * 100) * 10,
   616→          ) / 10
   617→        : 0;
   618→    return { total, qualified, avgScore, bestScore };
   619→  }, [exams]);
   620→
   621→  const selectedExam = useMemo(
   622→    () => exams.find((e) => e.id === selectedExamId) ?? null,
   623→    [exams, selectedExamId],
   624→  );
   625→
   626→  const openDetail = (examId: string) => {
   627→    setSelectedExamId(examId);
   628→    setDetailTab('overview');
   629→    setSheetOpen(true);
   630→  };
   631→
   632→  return (
   633→    <div className="space-y-6">
   634→      {/* Header */}
   635→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   636→        <div>
   637→          <h1 className="text-2xl font-bold tracking-tight text-foreground">
   638→            Exam History
   639→          </h1>
   640→          <p className="text-sm text-muted-foreground">
   641→            Browse and manage all your past exam results
   642→          </p>
   643→        </div>
   644→        <Button
   645→          onClick={() => setAddExamOpen(true)}
   646→          className="bg-emerald-600 text-white hover:bg-emerald-700"
   647→        >
   648→          <Plus className="mr-2 size-4" />
   649→          Add Exam
   650→        </Button>
   651→      </div>
   652→
   653→      {/* Stat Cards */}
   654→      <div className="grid grid-cols-2 gap-4 md:grid-cols-4">
   655→        <StatCard
   656→          icon={GraduationCap}
   657→          label="Total Exams"
   658→          value={stats.total}
   659→          colorClass="bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-300"
   660→        />
   661→        <StatCard
   662→          icon={Trophy}
   663→          label="Qualified"
   664→          value={stats.qualified}
   665→          change={
   666→            stats.total > 0
   667→              ? `${Math.round((stats.qualified / stats.total) * 100)}% rate`
   668→              : undefined
   669→          }
   670→          trend={stats.qualified >= stats.total * 0.5 ? 'up' : 'neutral'}
   671→        />
   672→        <StatCard
   673→          icon={BarChart3}
   674→          label="Avg Score"
   675→          value={`${stats.avgScore}%`}
   676→          colorClass="bg-teal-100 text-teal-700 dark:bg-teal-900/50 dark:text-teal-300"
   677→        />
   678→        <StatCard
   679→          icon={Star}
   680→          label="Best Score"
   681→          value={`${stats.bestScore}%`}
   682→          trend="up"
   683→          colorClass="bg-amber-100 text-amber-700 dark:bg-amber-900/50 dark:text-amber-300"
   684→        />
   685→      </div>
   686→
   687→      {/* Filter Bar */}
   688→      <Card>
   689→        <CardContent className="flex flex-wrap items-center gap-3 py-3">
   690→          <Filter className="size-4 text-muted-foreground" />
   691→          <Select value={category} onValueChange={(v) => { setCategory(v); setPage(1); }}>
   692→            <SelectTrigger className="w-[130px]">
   693→              <SelectValue placeholder="Category" />
   694→            </SelectTrigger>
   695→            <SelectContent>
   696→              <SelectItem value="all">All Categories</SelectItem>
   697→              <SelectItem value="Banking">Banking</SelectItem>
   698→              <SelectItem value="SSC">SSC</SelectItem>
   699→              <SelectItem value="Railway">Railway</SelectItem>
   700→              <SelectItem value="Insurance">Insurance</SelectItem>
   701→              <SelectItem value="Other">Other</SelectItem>
   702→            </SelectContent>
   703→          </Select>
   704→          <Select value={year} onValueChange={(v) => { setYear(v); setPage(1); }}>
   705→            <SelectTrigger className="w-[100px]">
   706→              <SelectValue placeholder="Year" />
   707→            </SelectTrigger>
   708→            <SelectContent>
   709→              <SelectItem value="all">All Years</SelectItem>
   710→              <SelectItem value="2025">2025</SelectItem>
   711→              <SelectItem value="2024">2024</SelectItem>
   712→              <SelectItem value="2023">2023</SelectItem>
   713→            </SelectContent>
   714→          </Select>
   715→          <Select value={status} onValueChange={(v) => { setStatus(v); setPage(1); }}>
   716→            <SelectTrigger className="w-[140px]">
   717→              <SelectValue placeholder="Status" />
   718→            </SelectTrigger>
   719→            <SelectContent>
   720→              <SelectItem value="all">All Status</SelectItem>
   721→              <SelectItem value="Qualified">Qualified</SelectItem>
   722→              <SelectItem value="Not Qualified">Not Qualified</SelectItem>
   723→              <SelectItem value="Awaited">Result Awaited</SelectItem>
   724→            </SelectContent>
   725→          </Select>
   726→          <Select value={sort} onValueChange={setSort}>
   727→            <SelectTrigger className="w-[160px]">
   728→              <SelectValue placeholder="Sort" />
   729→            </SelectTrigger>
   730→            <SelectContent>
   731→              <SelectItem value="examDate:desc">Newest First</SelectItem>
   732→              <SelectItem value="examDate:asc">Oldest First</SelectItem>
   733→              <SelectItem value="score:desc">Highest Score</SelectItem>
   734→              <SelectItem value="score:asc">Lowest Score</SelectItem>
   735→            </SelectContent>
   736→          </Select>
   737→        </CardContent>
   738→      </Card>
   739→
   740→      {/* Exam Cards */}
   741→      <div className="space-y-8">
   742→        {isLoading ? (
   743→          <div className="space-y-6">
   744→            <Skeleton className="h-8 w-24" />
   745→            {Array.from({ length: 3 }).map((_, i) => (
   746→              <Skeleton key={i} className="h-[120px] w-full rounded-xl" />
   747→            ))}
   748→          </div>
   749→        ) : Object.keys(grouped).length > 0 ? (
   750→          Object.entries(grouped).map(([yearLabel, yearExams]) => (
   751→            <div key={yearLabel}>
   752→              <h2 className="mb-3 text-lg font-semibold text-foreground">
   753→                {yearLabel}
   754→              </h2>
   755→              <div className="space-y-3">
   756→                {yearExams.map((exam) => {
   757→                  const gap = exam.cutoffGap;
   758→                  const gapColor =
   759→                    gap !== null && gap > 0
   760→                      ? 'text-red-600 dark:text-red-400'
   761→                      : gap !== null && gap <= 0
   762→                        ? 'text-emerald-600 dark:text-emerald-400'
   763→                        : 'text-muted-foreground';
   764→
   765→                  return (
   766→                    <Card
   767→                      key={exam.id}
   768→                      className="cursor-pointer transition-all hover:border-emerald-200 hover:shadow-md dark:hover:border-emerald-800"
   769→                      onClick={() => openDetail(exam.id)}
   770→                    >
   771→                      <CardContent className="flex flex-col gap-3 p-4 sm:flex-row sm:items-center sm:justify-between">
   772→                        {/* Left info */}
   773→                        <div className="flex items-start gap-3">
   774→                          <div className="flex flex-col items-center gap-1">
   775→                            <Badge variant="outline" className="text-xs">
   776→                              {exam.category}
   777→                            </Badge>
   778→                            <Badge
   779→                              variant="secondary"
   780→                              className="text-[10px]"
   781→                            >
   782→                              {exam.stage}
   783→                            </Badge>
   784→                          </div>
   785→                          <div>
   786→                            <p className="font-semibold text-foreground">
   787→                              {exam.name}
   788→                            </p>
   789→                            <p className="text-xs text-muted-foreground">
   790→                              {exam.org || 'Unknown'} ·{' '}
   791→                              {formatDate(exam.examDate)} · Attempt #
   792→                              {exam.attempt}
   793→                            </p>
   794→                          </div>
   795→                        </div>
   796→
   797→                        {/* Right stats */}
   798→                        <div className="flex items-center gap-4">
   799→                          <div className="text-right">
   800→                            <p className="text-lg font-bold">{exam.score}/{exam.maxScore}</p>
   801→                            <p className="text-xs text-muted-foreground">
   802→                              Cutoff: {exam.cutoff}
   803→                            </p>
   804→                          </div>
   805→                          <div className="text-right">
   806→                            <p className={`text-sm font-semibold ${gapColor}`}>
   807→                              {gap !== null ? (gap > 0 ? `+${gap}` : `${gap}`) : '—'}
   808→                            </p>
   809→                            <p className="text-[10px] text-muted-foreground">gap</p>
   810→                          </div>
   811→                          <Badge
   812→                            className={
   813→                              exam.result === 'Qualified'
   814→                                ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-300'
   815→                                : exam.result === 'Awaited'
   816→                                  ? 'bg-amber-100 text-amber-700 dark:bg-amber-900/50 dark:text-amber-300'
   817→                                  : 'bg-red-100 text-red-700 dark:bg-red-900/50 dark:text-red-300'
   818→                            }
   819→                          >
   820→                            {exam.result}
   821→                          </Badge>
   822→                          {exam.reflection && (
   823→                            <BookOpen className="size-4 text-emerald-600 dark:text-emerald-400" />
   824→                          )}
   825→                          <Eye className="size-4 text-muted-foreground" />
   826→                        </div>
   827→                      </CardContent>
   828→                    </Card>
   829→                  );
   830→                })}
   831→              </div>
   832→            </div>
   833→          ))
   834→        ) : (
   835→          <div className="flex flex-col items-center justify-center rounded-xl border border-dashed py-12 text-center">
   836→            <GraduationCap className="mb-3 size-10 text-muted-foreground/40" />
   837→            <p className="font-medium text-muted-foreground">
   838→              No exams found
   839→            </p>
   840→            <p className="mt-1 text-sm text-muted-foreground/60">
   841→              Try adjusting your filters or add your first exam.
   842→            </p>
   843→            <Button
   844→              variant="outline"
   845→              className="mt-4"
   846→              onClick={() => setAddExamOpen(true)}
   847→            >
   848→              <Plus className="mr-2 size-4" />
   849→              Add Exam
   850→            </Button>
   851→          </div>
   852→        )}
   853→      </div>
   854→
   855→      {/* Load More */}
   856→      {hasMore && (
   857→        <div className="flex justify-center">
   858→          <Button
   859→            variant="outline"
   860→            onClick={() => setPage((p) => p + 1)}
   861→            disabled={isFetching}
   862→          >
   863→            {isFetching ? (
   864→              <Loader2 className="mr-2 size-4 animate-spin" />
   865→            ) : null}
   866→            Load More
   867→          </Button>
   868→        </div>
   869→      )}
   870→
   871→      {/* Detail Side Sheet (Mobile) */}
   872→      <Sheet open={sheetOpen} onOpenChange={setSheetOpen}>
   873→        <SheetContent className="w-full overflow-y-auto sm:max-w-md">
   874→          {selectedExam && (
   875→            <>
   876→              <SheetHeader>
   877→                <SheetTitle className="text-lg">{selectedExam.name}</SheetTitle>
   878→              </SheetHeader>
   879→              <Tabs value={detailTab} onValueChange={setDetailTab} className="mt-4">
   880→                <TabsList className="grid w-full grid-cols-2">
   881→                  <TabsTrigger value="overview">Overview</TabsTrigger>
   882→                  <TabsTrigger value="reflection">Reflection</TabsTrigger>
   883→                </TabsList>
   884→                <TabsContent value="overview">
   885→                  <ExamDetailContent exam={selectedExam} activeTab="overview" />
   886→                </TabsContent>
   887→                <TabsContent value="reflection">
   888→                  <ExamDetailContent exam={selectedExam} activeTab="reflection" />
   889→                </TabsContent>
   890→              </Tabs>
   891→            </>
   892→          )}
   893→        </SheetContent>
   894→      </Sheet>
   895→
   896→      {/* Add Exam Dialog */}
   897→      <AddExamDialog open={addExamOpen} onOpenChange={setAddExamOpen} />
   898→    </div>
   899→  );
   900→}
