    42→/* ─── Heatmap Cell Color Logic (gradient) ──────────────────── */
    43→function getHeatmapCellClasses(value: number): string {
    44→  if (value <= 20) return 'bg-red-500/20 text-red-700 dark:bg-red-500/30 dark:text-red-300';
    45→  if (value <= 30) return 'bg-red-400/15 text-red-600 dark:bg-red-400/25 dark:text-red-400';
    46→  if (value <= 40) return 'bg-orange-400/15 text-orange-700 dark:bg-orange-400/25 dark:text-orange-300';
    47→  if (value <= 50) return 'bg-amber-400/15 text-amber-700 dark:bg-amber-400/25 dark:text-amber-300';
    48→  if (value <= 60) return 'bg-yellow-400/15 text-yellow-700 dark:bg-yellow-400/25 dark:text-yellow-300';
    49→  if (value <= 70) return 'bg-lime-400/15 text-lime-700 dark:bg-lime-400/25 dark:text-lime-300';
    50→  if (value <= 80) return 'bg-emerald-400/15 text-emerald-700 dark:bg-emerald-400/25 dark:text-emerald-300';
    51→  return 'bg-emerald-500/20 text-emerald-600 dark:bg-emerald-500/30 dark:text-emerald-400';
    52→}
    53→
    54→function getHeatmapDotColor(value: number): string {
    55→  if (value <= 30) return 'bg-red-500';
    56→  if (value <= 50) return 'bg-amber-500';
    57→  if (value <= 70) return 'bg-yellow-500';
    58→  return 'bg-emerald-500';
    59→}

    61→function getSeverityBadge(value: number) {
    62→  if (value <= 30) return { label: 'Critical', cls: 'bg-red-100 text-red-700 dark:bg-red-900/50 dark:text-red-300 border-red-200 dark:border-red-800' };
    63→  if (value <= 50) return { label: 'Moderate', cls: 'bg-amber-100 text-amber-700 dark:bg-amber-900/50 dark:text-amber-300 border-amber-200 dark:border-amber-800' };
    64→  return { label: 'Minor', cls: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-300 border-emerald-200 dark:border-emerald-800' };
    65→}
