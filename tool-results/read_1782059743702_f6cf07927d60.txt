    68→    68→function formatDate(dateStr: string): string {
    69→    69→  const d = new Date(dateStr);
    70→    70→  return d.toLocaleDateString('en-IN', {
    71→    71→    day: 'numeric',
    72→    72→    month: 'short',
    73→    73→    year: 'numeric',
    74→    74→  });
    75→    75→}

    77→    77→function getCountdownLabel(days: number | null): {
    78→    78→  label: string;
    79→    79→  className: string;
    80→    80→} {
    81→    81→  if (days === null || days === undefined)
    82→    82→    return { label: '—', className: 'text-muted-foreground' };
    83→    83→  if (days === 0) return { label: 'Today!', className: 'text-red-600 font-bold' };
    84→    84→  if (days === 1) return { label: 'Tomorrow', className: 'text-amber-600 font-semibold' };
    85→    85→  if (days <= 7) return { label: `${days} days`, className: 'text-red-600 font-semibold' };
    86→    86→  if (days <= 30) return { label: `${days} days`, className: 'text-amber-600 font-medium' };
    87→    87→  return { label: `${days} days`, className: 'text-emerald-600' };
    88→    88→}

   145→   145→function UpcomingExamDialog({
   146→   146→  open,
   147→   147→  onOpenChange,
   148→   148→  editExam,
   149→   149→}: {
   150→   150→  open: boolean;
   151→   151→  onOpenChange: (open: boolean) => void;
   152→   152→  editExam: UpcomingExam | null;
   153→   153→}) {
   154→   154→  const queryClient = useQueryClient();
   155→   155→  const isEdit = editExam !== null;
   156→   156→
   157→   157→  const [form, setForm] = useState(() =>
   158→   158→    editExam
   159→   159→      ? {
   160→   160→          name: editExam.name,
   161→   161→          org: editExam.org ?? '',
   162→   162→          priority: editExam.priority,
   163→   163→          examDate: new Date(editExam.examDate).toISOString().split('T')[0],
   164→   164→        }
   165→   165→      : { name: '', org: '', priority: 'Medium', examDate: '' },
   166→   166→  );
   167→   167→
   168→   168→  const createMutation = useMutation({
   169→   169→    mutationFn: () =>
   170→   170→      createUpcomingExam({
   171→   171→        name: form.name,
   172→   172→        org: form.org || undefined,
   173→   173→        priority: form.priority,
   174→   174→        examDate: form.examDate,
   175→   175→      }),
   176→   176→    onSuccess: () => {
   177→   177→      toast.success(
   178→   178→        isEdit ? 'Exam updated successfully!' : 'Upcoming exam added!',
   179→   179→      );
   180→   180→      queryClient.invalidateQueries({ queryKey: ['upcoming-exams'] });
   181→   181→      onOpenChange(false);
   182→   182→    },
   183→   183→    onError: () => {
   184→   184→      toast.error('Failed to save exam. Please try again.');
   185→   185→    },
   186→   186→  });
   187→   187→
   188→   188→  const updateMutation = useMutation({
   189→   189→    mutationFn: () =>
   190→   190→      updateUpcomingExam(editExam!.id, {
   191→   191→        name: form.name,
   192→   192→        org: form.org || undefined,
   193→   193→        priority: form.priority,
   194→   194→        examDate: form.examDate,
   195→   195→      }),
   196→   196→    onSuccess: () => {
   197→   197→      toast.success('Exam updated successfully!');
   198→   198→      queryClient.invalidateQueries({ queryKey: ['upcoming-exams'] });
   199→   199→      onOpenChange(false);
   200→   200→    },
   201→   201→    onError: () => {
   202→   202→      toast.error('Failed to update exam. Please try again.');
   203→   203→    },
   204→   204→  });
   205→   205→
   206→   206→  const handleSubmit = () => {
   207→   207→    if (isEdit) {
   208→   208→      updateMutation.mutate();
   209→   209→    } else {
   210→   210→      createMutation.mutate();
   211→   211→    }
   212→   212→  };
   213→   213→
   214→   214→  const isPending = createMutation.isPending || updateMutation.isPending;
   215→   215→
   216→   216→  return (
   217→   217→    <Dialog open={open} onOpenChange={onOpenChange}>
   218→   218→      <DialogContent className="sm:max-w-md">
   219→   219→        <DialogHeader>
   220→   220→          <DialogTitle>
   221→   221→            {isEdit ? 'Edit Upcoming Exam' : 'Add Upcoming Exam'}
   222→   222→          </DialogTitle>
   223→   223→        </DialogHeader>
   224→   224→        <div className="grid gap-4 py-2">
   225→   225→          <div className="grid gap-2">
   226→   226→            <Label htmlFor="ue-name">Exam Name</Label>
   227→   227→            <Input
   228→   228→              id="ue-name"
   229→   229→              placeholder="e.g. IBPS PO 2025"
   230→   230→              value={form.name}
   231→   231→              onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   232→   232→            />
   233→   233→          </div>
   234→   234→          <div className="grid gap-2">
   235→   235→            <Label htmlFor="ue-org">Organization (optional)</Label>
   236→   236→            <Input
   237→   237→              id="ue-org"
   238→   238→              placeholder="e.g. IBPS"
   239→   239→              value={form.org}
   240→   240→              onChange={(e) => setForm((f) => ({ ...f, org: e.target.value }))}
   241→   241→            />
   242→   242→          </div>
   243→   243→          <div className="grid gap-2">
   244→   244→            <Label>Priority</Label>
   245→   245→            <Select
   246→   246→              value={form.priority}
   247→   247→              onValueChange={(v) => setForm((f) => ({ ...f, priority: v }))}
   248→   248→            >
   249→   249→              <SelectTrigger>
   250→   250→                <SelectValue />
   251→   251→              </SelectTrigger>
   252→   252→              <SelectContent>
   253→   253→                <SelectItem value="High">High</SelectItem>
   254→   254→                <SelectItem value="Medium">Medium</SelectItem>
   255→   255→                <SelectItem value="Low">Low</SelectItem>
   256→   256→              </SelectContent>
   257→   257→            </Select>
   258→   258→          </div>
   259→   259→          <div className="grid gap-2">
   260→   260→            <Label htmlFor="ue-date">Exam Date</Label>
   261→   261→            <Input
   262→   262→              id="ue-date"
   263→   263→              type="date"
   264→   264→              value={form.examDate}
   265→   265→              onChange={(e) => setForm((f) => ({ ...f, examDate: e.target.value }))}
   266→   266→            />
   267→   267→          </div>
   268→   268→        </div>
   269→   269→        <DialogFooter>
   270→   270→          <DialogClose asChild>
   271→   271→            <Button variant="outline">Cancel</Button>
   272→   272→          </DialogClose>
   273→   273→          <Button
   274→   274→            onClick={handleSubmit}
   275→   275→            disabled={isPending || !form.name || !form.examDate}
   276→   276→            className="bg-emerald-600 text-white hover:bg-emerald-700"
   277→   277→          >
   278→   278→            {isPending ? (
   279→   279→              <Loader2 className="mr-2 size-4 animate-spin" />
   280→   280→            ) : null}
   281→   281→            {isEdit ? 'Update' : 'Add Exam'}
   282→   282→          </Button>
   283→   283→        </DialogFooter>
   284→   284→      </DialogContent>
   285→   285→    </Dialog>
   286→   286→  );
   287→   287→}
