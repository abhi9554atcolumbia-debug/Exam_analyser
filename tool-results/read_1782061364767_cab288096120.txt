    67→    67→function getGreeting(): string {
    68→    68→  const h = new Date().getHours();
    69→    69→  if (h < 12) return 'Good morning';
    70→    70→  if (h < 17) return 'Good afternoon';
    71→    71→  return 'Good evening';
    72→    72→}

    74→    74→function formatDate(dateStr: string): string {
    75→    75→  const d = new Date(dateStr);
    76→    76→  return d.toLocaleDateString('en-IN', {
    77→    77→    day: 'numeric',
    78→    78→    month: 'short',
    79→    79→    year: 'numeric',
    80→    80→  });
    81→    81→}
    82→    82→
    83→    83→function formatCountdown(days: number | null): string {
    84→    84→  if (days === null || days === undefined) return '—';
    85→    85→  if (days === 0) return 'Today!';
    86→    86→  if (days === 1) return 'Tomorrow';
    87→    87→  return `${days} days left`;
    88→    88→}
