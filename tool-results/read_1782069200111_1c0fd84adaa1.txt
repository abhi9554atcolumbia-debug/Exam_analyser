    46→    46→  if (value <= 40) return 'bg-orange-400/15 text-orange-700 dark:bg-orange-400/25 dark:text-orange-300';
    47→    47→  if (value <= 50) return 'bg-amber-400/15 text-amber-700 dark:bg-amber-400/25 dark:text-amber-300';
    48→    48→  if (value <= 60) return 'bg-yellow-400/15 text-yellow-700 dark:bg-yellow-400/25 dark:text-yellow-300';
    49→    49→  if (value <= 70) return 'bg-lime-400/15 text-lime-700 dark:bg-lime-400/25 dark:text-lime-300';
