    67→    67→function formatDate(dateStr: string): string {
    68→    68→  const d = new Date(dateStr);
    69→    69→  return d.toLocaleDateString('en-IN', {
    70→    70→    day: 'numeric',
    71→    71→    month: 'short',
    72→    72→    year: 'numeric',
    73→    73→  });

    76→    76→function groupExamsByYear(exams: Exam[]): Record<string, Exam[]> {
    77→    77→  const groups: Record<string, Exam[]> = {};
    78→    78→  for (const exam of exams) {
    79→    79→    const year = new Date(exam.examDate).getFullYear().toString();
    80→    80→    if (!groups[year]) groups[year] = [];
    81→    81→    groups[year].push(exam);
    82→    82→  }
    83→    83→  return Object.fromEntries(
    84→    84→    Object.entries(groups).sort(([a], [b]) => Number(b) - Number(a)),
    85→    85→  );
    86→    86→}
