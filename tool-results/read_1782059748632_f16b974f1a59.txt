    68→function formatDate(dateStr: string): string {
    69→  const d = new Date(dateStr);
    70→  return d.toLocaleDateString('en-IN', {
    71→    day: 'numeric',
    72→    month: 'short',
    73→    year: 'numeric',
    74→  });
    75→}
    76→
    77→function getCountdownLabel(days: number | null): {
    78→  label: string;
    79→  className: string;
    80→} {
    81→  if (days === null || days === undefined)
    82→    return { label: '—', className: 'text-muted-foreground' };
    83→  if (days === 0) return { label: 'Today!', className: 'text-red-600 font-bold' };
    84→  if (days === 1) return { label: 'Tomorrow', className: 'text-amber-600 font-semibold' };
    85→  if (days <= 7) return { label: `${days} days`, className: 'text-red-600 font-semibold' };
    86→  if (days <= 30) return { label: `${days} days`, className: 'text-amber-600 font-medium' };
    87→  return { label: `${days} days`, className: 'text-emerald-600' };
    88→}

    90→function getPriorityConfig(priority: string) {
    91→  switch (priority) {
    92→    case 'High':
    93→      return {
    94→        borderColor: 'border-l-red-500',
    95→        bgClass: 'bg-red-100 text-red-700 dark:bg-red-900/50 dark:text-red-300',
    96→        badgeClass: 'bg-red-100 text-red-700 dark:bg-red-900/50 dark:text-red-300',
    97→      };
    98→    case 'Medium':
    99→      return {
   100→        borderColor: 'border-l-amber-500',
   101→        bgClass: 'bg-amber-100 text-amber-700 dark:bg-amber-900/50 dark:text-amber-300',
   102→        badgeClass: 'bg-amber-100 text-amber-700 dark:bg-amber-900/50 dark:text-amber-300',
   103→      };
   104→    case 'Low':
   105→      return {
   106→        borderColor: 'border-l-emerald-500',
   107→        bgClass: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-300',
   108→        badgeClass: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-300',
   109→      };
   110→    default:
   111→      return {
   112→        borderColor: 'border-l-muted',
   113→        bgClass: 'bg-muted text-muted-foreground',
   114→        badgeClass: 'bg-muted text-muted-foreground',
   115→      };
   116→  }
   117→}

   120→  switch (appStatus) {
   121→    case 'completed':

   128→      return (

   130→          In Progress

   134→      return (

   144→// ─── Add/Edit Upcoming Exam Dialog ─────────────────────────
   145→function UpcomingExamDialog({
   146→  open,
   147→  onOpenChange,
   148→  editExam,
   149→}: {
   150→  open: boolean;
   151→  onOpenChange: (open: boolean) => void;
   152→  editExam: UpcomingExam | null;
   153→}) {
   154→  const queryClient = useQueryClient();
   155→  const isEdit = editExam !== null;
   156→
   157→  const [form, setForm] = useState(() =>
   158→    editExam
   159→      ? {
   160→          name: editExam.name,
   161→          org: editExam.org ?? '',
   162→          priority: editExam.priority,
   163→          examDate: new Date(editExam.examDate).toISOString().split('T')[0],
   164→        }
   165→      : { name: '', org: '', priority: 'Medium', examDate: '' },
   166→  );
   167→
   168→  const createMutation = useMutation({
   169→    mutationFn: () =>
   170→      createUpcomingExam({
   171→        name: form.name,
   172→        org: form.org || undefined,
   173→        priority: form.priority,
   174→        examDate: form.examDate,
   175→      }),
   176→    onSuccess: () => {
   177→      toast.success(
   178→        isEdit ? 'Exam updated successfully!' : 'Upcoming exam added!',
   179→      );
   180→      queryClient.invalidateQueries({ queryKey: ['upcoming-exams'] });
   181→      onOpenChange(false);
   182→    },
   183→    onError: () => {
   184→      toast.error('Failed to save exam. Please try again.');
   185→    },
   186→  });
   187→
   188→  const updateMutation = useMutation({
   189→    mutationFn: () =>
   190→      updateUpcomingExam(editExam!.id, {
   191→        name: form.name,
   192→        org: form.org || undefined,
   193→        priority: form.priority,
   194→        examDate: form.examDate,
   195→      }),
   196→    onSuccess: () => {
   197→      toast.success('Exam updated successfully!');
   198→      queryClient.invalidateQueries({ queryKey: ['upcoming-exams'] });
   199→      onOpenChange(false);
   200→    },
   201→    onError: () => {
   202→      toast.error('Failed to update exam. Please try again.');
   203→    },
   204→  });
   205→
   206→  const handleSubmit = () => {
   207→    if (isEdit) {
   208→      updateMutation.mutate();
   209→    } else {
   210→      createMutation.mutate();
   211→    }
   212→  };
   213→
   214→  const isPending = createMutation.isPending || updateMutation.isPending;
   215→
   216→  return (
   217→    <Dialog open={open} onOpenChange={onOpenChange}>
   218→      <DialogContent className="sm:max-w-md">
   219→        <DialogHeader>
   220→          <DialogTitle>
   221→            {isEdit ? 'Edit Upcoming Exam' : 'Add Upcoming Exam'}
   222→          </DialogTitle>
   223→        </DialogHeader>
   224→        <div className="grid gap-4 py-2">
   225→          <div className="grid gap-2">
   226→            <Label htmlFor="ue-name">Exam Name</Label>
   227→            <Input
   228→              id="ue-name"
   229→              placeholder="e.g. IBPS PO 2025"
   230→              value={form.name}
   231→              onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))}
   232→            />
   233→          </div>
   234→          <div className="grid gap-2">
   235→            <Label htmlFor="ue-org">Organization (optional)</Label>
   236→            <Input
   237→              id="ue-org"
   238→              placeholder="e.g. IBPS"
   239→              value={form.org}
   240→              onChange={(e) => setForm((f) => ({ ...f, org: e.target.value }))}
   241→            />
   242→          </div>
   243→          <div className="grid gap-2">
   244→            <Label>Priority</Label>
   245→            <Select
   246→              value={form.priority}
   247→              onValueChange={(v) => setForm((f) => ({ ...f, priority: v }))}
   248→            >
   249→              <SelectTrigger>
   250→                <SelectValue />
   251→              </SelectTrigger>
   252→              <SelectContent>
   253→                <SelectItem value="High">High</SelectItem>
   254→                <SelectItem value="Medium">Medium</SelectItem>
   255→                <SelectItem value="Low">Low</SelectItem>
   256→              </SelectContent>
   257→            </Select>
   258→          </div>
   259→          <div className="grid gap-2">
   260→            <Label htmlFor="ue-date">Exam Date</Label>
   261→            <Input
   262→              id="ue-date"
   263→              type="date"
   264→              value={form.examDate}
   265→              onChange={(e) => setForm((f) => ({ ...f, examDate: e.target.value }))}
   266→            />
   267→          </div>
   268→        </div>
   269→        <DialogFooter>
   270→          <DialogClose asChild>
   271→            <Button variant="outline">Cancel</Button>
   272→          </DialogClose>
   273→          <Button
   274→            onClick={handleSubmit}
   275→            disabled={isPending || !form.name || !form.examDate}
   276→            className="bg-emerald-600 text-white hover:bg-emerald-700"
   277→          >
   278→            {isPending ? (
   279→              <Loader2 className="mr-2 size-4 animate-spin" />
   280→            ) : null}
   281→            {isEdit ? 'Update' : 'Add Exam'}
   282→          </Button>
   283→        </DialogFooter>
   284→      </DialogContent>
   285→    </Dialog>
   286→  );
   287→}
