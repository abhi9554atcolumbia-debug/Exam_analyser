   127→   127→function formatFileSize(bytes: number | null): string {
   128→   128→  if (bytes == null) return '—';
   129→   129→  if (bytes === 0) return '0 B';
   130→   130→  const units = ['B', 'KB', 'MB', 'GB'];
   131→   131→  const k = 1024;
   132→   132→  const i = Math.floor(Math.log(bytes) / Math.log(k));
   133→   133→  const size = bytes / Math.pow(k, i);
   134→   134→  return `${size.toFixed(i === 0 ? 0 : 1)} ${units[i]}`;
   135→   135→}

   137→   137→function formatDate(dateStr: string): string {
   138→   138→  try {
   139→   139→    return new Date(dateStr).toLocaleDateString('en-IN', {
   140→   140→      day: '2-digit',
   141→   141→      month: 'short',
   142→   142→      year: 'numeric',
   143→   143→    });
   144→   144→  } catch {
   145→   145→    return dateStr;
   146→   146→  }
   147→   147→}
