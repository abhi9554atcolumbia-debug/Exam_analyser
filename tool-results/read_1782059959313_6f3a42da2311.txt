    65→// ─── Helpers ───────────────────────────────────────────────
    66→function getGreeting(): string {
    67→  const h = new Date().getHours();
    68→  if (h < 12) return 'Good morning';
    69→  if (h < 17) return 'Good afternoon';
    70→  return 'Good evening';
    71→}
    72→
    73→function formatDate(dateStr: string): string {
    74→  const d = new Date(dateStr);
    75→  return d.toLocaleDateString('en-IN', {
    76→    day: 'numeric',
    77→    month: 'short',
    78→    year: 'numeric',
    79→  });
    80→}
    81→
    82→function formatCountdown(days: number | null): string {
    83→  if (days === null || days === undefined) return '—';
    84→  if (days === 0) return 'Today!';
    85→  if (days === 1) return 'Tomorrow';
    86→  return `${days} days left`;
    87→}
