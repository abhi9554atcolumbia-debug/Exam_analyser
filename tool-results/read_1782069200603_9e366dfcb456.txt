    51→    51→function filterByPeriod(exams: Exam[], period: string): Exam[] {
    52→    52→  if (period === 'all') return exams;
    53→    53→  const now = new Date();
    54→    54→  const monthsMap: Record<string, number> = { '3m': 3, '6m': 6, '1y': 12 };
    55→    55→  const months = monthsMap[period] ?? 999;
    56→    56→  const cutoff = new Date(now.getFullYear(), now.getMonth() - months, now.getDate());
    57→    57→  return exams.filter((e) => new Date(e.examDate) >= cutoff);
    58→    58→}
