    55→function getDueCountdown(dueDate: string | null): { text: string; urgent: boolean } | null {
    56→  if (!dueDate) return null;
    57→  const days = differenceInDays(parseISO(dueDate), new Date());
    58→  if (days < 0) return { text: `Overdue by ${Math.abs(days)} day${Math.abs(days) !== 1 ? 's' : ''}`, urgent: true };
    59→  if (days === 0) return { text: 'Due today', urgent: true };
    60→  if (days === 1) return { text: 'Due tomorrow', urgent: true };
    61→  if (days <= 7) return { text: `Due in ${days} days`, urgent: true };
    62→  return { text: `Due in ${days} days`, urgent: false };
    63→}

    84→/* ── Completion Rate Ring ──────────────────────────────────── */
    85→function CompletionRateRing({ rate, total }: { rate: number; total: number }) {
    86→  const radius = 40;
    87→  const circumference = 2 * Math.PI * radius;
    88→  const offset = circumference - (rate / 100) * circumference;
    89→
    90→  return (
    91→    <div className="flex flex-col items-center justify-center">
    92→      <div className="relative">
    93→        <svg width="110" height="110" className="-rotate-90">
    94→          <circle
    95→            cx="55" cy="55" r={radius}
    96→            fill="none"
    97→            strokeWidth="8"
    98→            className="stroke-muted"
    99→          />
   100→          <circle
   101→            cx="55" cy="55" r={radius}
   102→            fill="none"
   103→            strokeWidth="8"
   104→            strokeLinecap="round"
   105→            className="stroke-emerald-500 transition-all duration-700"
   106→            strokeDasharray={circumference}
   107→            strokeDashoffset={offset}
   108→          />
   109→        </svg>
   110→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   111→          <span className="text-2xl font-extrabold text-foreground">{rate}%</span>
   112→          <span className="text-[10px] text-muted-foreground">completed</span>
   113→        </div>
   114→      </div>
   115→      <p className="mt-2 text-xs text-muted-foreground">{total} total goals</p>
   116→    </div>
   117→  );
   118→}
