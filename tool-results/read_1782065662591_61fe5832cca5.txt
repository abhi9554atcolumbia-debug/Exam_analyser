   319→              </Badge>
   320→            ))}

   328→// ── Create Reflection Dialog ─────────────────────────────────
   329→function CreateReflectionDialog({
   330→  open,
   331→  onOpenChange,
   332→}: {
   333→  open: boolean;
   334→  onOpenChange: (open: boolean) => void;
   335→}) {
   336→  const queryClient = useQueryClient();
   337→  const { navigate } = useNavigationStore();
   338→
   339→  const [examId, setExamId] = useState('');
   340→  const [difficulty, setDifficulty] = useState('');
   341→  const [confidence, setConfidence] = useState(50);
   342→  const [emotionalState, setEmotionalState] = useState('');
   343→  const [whatWentWell, setWhatWentWell] = useState('');
   344→  const [whatWentWrong, setWhatWentWrong] = useState('');
   345→  const [biggestLesson, setBiggestLesson] = useState('');
   346→  const [actionPlan, setActionPlan] = useState('');
   347→  const [targetScore, setTargetScore] = useState('');
   348→
   349→  const { data: examOptions, isLoading: loadingOptions } = useQuery({
   350→    queryKey: ['reflection-exam-options'],
   351→    queryFn: getReflectionExamOptions,
   352→  });
   353→
   354→  const { data: selectedExam, isLoading: loadingExam } = useQuery({
   355→    queryKey: ['exam-for-dialog', examId],
   356→    queryFn: () => getExam(examId),
   357→    enabled: !!examId,
   358→  });
   359→
   360→  const createMutation = useMutation({
   361→    mutationFn: createReflection,
   362→    onSuccess: () => {
   363→      toast.success('Reflection created successfully!');
   364→      queryClient.invalidateQueries({ queryKey: ['reflections'] });
   365→      queryClient.invalidateQueries({ queryKey: ['reflection-exam-options'] });
   366→      queryClient.invalidateQueries({ queryKey: ['analytics'] });
   367→      queryClient.invalidateQueries({ queryKey: ['weakness'] });
   368→      onOpenChange(false);
   369→      resetForm();
   370→    },
   371→    onError: (err: Error) => {
   372→      toast.error(err.message || 'Failed to create reflection');
   373→    },
   374→  });
   375→
   376→  const resetForm = () => {
   377→    setExamId('');
   378→    setDifficulty('');
   379→    setConfidence(50);
   380→    setEmotionalState('');
   381→    setWhatWentWell('');
   382→    setWhatWentWrong('');
   383→    setBiggestLesson('');
   384→    setActionPlan('');
   385→    setTargetScore('');
   386→  };
   387→
   388→  const handleSubmit = () => {
   389→    if (!examId) {
   390→      toast.error('Please select an exam');
   391→      return;
   392→    }
   393→    createMutation.mutate({
   394→      examId,
   395→      examName: selectedExam?.name ?? '',
   396→      examDate: selectedExam?.examDate ?? null,
   397→      score: selectedExam?.score ?? null,
   398→      cutoff: selectedExam?.cutoff ?? null,
   399→      gap: selectedExam?.cutoffGap ?? null,
   400→      result: selectedExam?.result ?? null,
   401→      difficulty: difficulty || null,
   402→      confidence,
   403→      emotionalState: emotionalState || null,
   404→      whatWentWrong: whatWentWrong || null,
   405→      whatWentWell: whatWentWell || null,
   406→      biggestLesson: biggestLesson || null,
   407→      actionPlan: actionPlan || null,
   408→      targetScore: targetScore ? Number(targetScore) : null,
   409→    });
   410→  };
   411→
   412→  const ToggleButton = ({
   413→    label,
   414→    active,
   415→    onClick,
   416→    activeColor,
   417→  }: {
   418→    label: string;
   419→    active: boolean;
   420→    onClick: () => void;
   421→    activeColor?: string;
   422→  }) => (
   423→    <button
   424→      type="button"
   425→      onClick={onClick}
   426→      className={cn(
   427→        'rounded-full border px-3 py-1.5 text-xs font-medium transition-all duration-200',
   428→        active
   429→          ? activeColor
   430→            ? activeColor
   431→            : 'border-emerald-500 bg-emerald-500/15 text-emerald-700 dark:text-emerald-400'
   432→          : 'border-border bg-background text-muted-foreground hover:border-emerald-400 hover:text-emerald-600',
   433→      )}
   434→    >
   435→      {label}
   436→    </button>
   437→  );
   438→
   439→  return (
   440→    <Dialog open={open} onOpenChange={(v) => { onOpenChange(v); if (!v) resetForm(); }}>
   441→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-lg">
   442→        <DialogHeader>
   443→          <DialogTitle className="flex items-center gap-2">
   444→            <PenLine className="h-5 w-5 text-emerald-600" />
   445→            Create Reflection
   446→          </DialogTitle>
   447→          <DialogDescription>
   448→            Write a detailed reflection for your exam to track progress
   449→          </DialogDescription>
   450→        </DialogHeader>
   451→
   452→        <div className="space-y-5 mt-2">
   453→          {/* Exam Selection */}
   454→          <div className="space-y-2">
   455→            <Label className="text-sm font-medium">
   456→              Select Exam <span className="text-red-500">*</span>
   457→            </Label>
   458→            <Select value={examId} onValueChange={setExamId}>
   459→              <SelectTrigger>
   460→                <SelectValue placeholder="Choose an exam..." />
   461→              </SelectTrigger>
   462→              <SelectContent>
   463→                {examOptions?.map((e) => (
   464→                  <SelectItem key={e.id} value={e.id}>
   465→                    {e.name} — {e.examDate}
   466→                  </SelectItem>
   467→                ))}
   468→                {(!examOptions || examOptions.length === 0) && !loadingOptions && (
   469→                  <SelectItem value="__none" disabled>
   470→                    No exams available
   471→                  </SelectItem>
   472→                )}
   473→              </SelectContent>
   474→            </Select>
   475→          </div>
   476→
   477→          {/* Exam Summary */}
   478→          {loadingExam && examId && (
   479→            <div className="rounded-lg border border-emerald-200 bg-emerald-50/50 p-3 dark:border-emerald-800 dark:bg-emerald-950/20">
   480→              <Skeleton className="h-4 w-40" />
   481→              <Skeleton className="mt-2 h-3 w-24" />
   482→            </div>
   483→          )}
   484→          {selectedExam && (
   485→            <div className="rounded-lg border border-emerald-200 bg-emerald-50/50 p-3 dark:border-emerald-800 dark:bg-emerald-950/20">
   486→              <p className="text-sm font-semibold text-foreground">{selectedExam.name}</p>
   487→              <p className="text-xs text-muted-foreground">
   488→                {selectedExam.examDate} · {selectedExam.category} · Score: {selectedExam.score}/{selectedExam.maxScore}
   489→                {selectedExam.result && (
   490→                  <span className={cn(
   491→                    'ml-1 font-medium',
   492→                    selectedExam.result.toLowerCase().includes('qualified')
   493→                      ? 'text-emerald-600 dark:text-emerald-400'
   494→                      : 'text-red-600 dark:text-red-400',
   495→                  )}>
   496→                    · {selectedExam.result}
   497→                  </span>
   498→                )}
   499→              </p>
   500→            </div>
   501→          )}
   502→
   503→          <Separator />
   504→
   505→          {/* Difficulty */}
   506→          <div className="space-y-2">
   507→            <Label className="text-sm font-medium">Difficulty Level</Label>
   508→            <div className="flex flex-wrap gap-2">
   509→              {DIFFICULTY_OPTIONS.map((d) => (
   510→                <ToggleButton
   511→                  key={d}
   512→                  label={d}
   513→                  active={difficulty === d}
   514→                  onClick={() => setDifficulty(difficulty === d ? '' : d)}
   515→                  activeColor={
   516→                    d === 'Easy' ? 'border-emerald-500 bg-emerald-500/15 text-emerald-700 dark:text-emerald-400'
   517→                    : d === 'Moderate' ? 'border-amber-500 bg-amber-500/15 text-amber-700 dark:text-amber-400'
   518→                    : d === 'Hard' ? 'border-orange-500 bg-orange-500/15 text-orange-700 dark:text-orange-400'
   519→                    : 'border-red-500 bg-red-500/15 text-red-700 dark:text-red-400'
   520→                  }
   521→                />
   522→              ))}
   523→            </div>
   524→          </div>
   525→
   526→          {/* Confidence Slider */}
   527→          <div className="space-y-3">
   528→            <div className="flex items-center justify-between">
   529→              <Label className="text-sm font-medium">Confidence Level</Label>
   530→              <span className="text-sm font-bold text-emerald-600 dark:text-emerald-400">{confidence}%</span>
   531→            </div>
   532→            <Slider
   533→              value={[confidence]}
   534→              onValueChange={(v) => setConfidence(v[0])}
   535→              min={0}
